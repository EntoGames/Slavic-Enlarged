namespace = wsl_debug


wsl_debug.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "============================================================"
		debug_log = "=== WSL FAITH DIST: START (date=[GetYear].[GetMonth]) ==="
		debug_log = "============================================================"

		set_variable = { name = wsl_dbg_base    value = 0 }
		set_variable = { name = wsl_dbg_east    value = 0 }
		set_variable = { name = wsl_dbg_west    value = 0 }
		set_variable = { name = wsl_dbg_south   value = 0 }
		set_variable = { name = wsl_dbg_unified value = 0 }

		every_county = {
			limit = { faith = faith:wsl_slavic_base }
			root = { change_variable = { name = wsl_dbg_base add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_east }
			root = { change_variable = { name = wsl_dbg_east add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_west }
			root = { change_variable = { name = wsl_dbg_west add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_south }
			root = { change_variable = { name = wsl_dbg_south add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_unified }
			root = { change_variable = { name = wsl_dbg_unified add = 1 } }
		}

		debug_log = "  wsl_slavic_base    : [root.MakeScope.Var('wsl_dbg_base').GetValue|0] counties"
		debug_log = "  wsl_slavic_east    : [root.MakeScope.Var('wsl_dbg_east').GetValue|0] counties"
		debug_log = "  wsl_slavic_west    : [root.MakeScope.Var('wsl_dbg_west').GetValue|0] counties"
		debug_log = "  wsl_slavic_south   : [root.MakeScope.Var('wsl_dbg_south').GetValue|0] counties"
		debug_log = "  wsl_slavic_unified : [root.MakeScope.Var('wsl_dbg_unified').GetValue|0] counties"
		debug_log = "============================================================"
		debug_log = "=== WSL FAITH DIST: END ==="
		debug_log = "============================================================"

		remove_variable = wsl_dbg_base
		remove_variable = wsl_dbg_east
		remove_variable = wsl_dbg_west
		remove_variable = wsl_dbg_south
		remove_variable = wsl_dbg_unified
	}

	option = { name = wsl_debug_ok }
}

wsl_debug.0004 = {
	type = character_event
	hidden = yes

	immediate = {
		debug_log = "=== WSL REFORM CHECK: [ROOT.GetName] ([GetYear].[GetMonth]) ==="

		if = {
			limit = { highest_held_title_tier >= tier_kingdom }
			debug_log = "  [PASS] Tier: >= Kingdom"
		}
		else = {
			debug_log = "  [FAIL] Tier: need Kingdom+"
		}

		if = {
			limit = {
				OR = {
					faith = faith:wsl_slavic_east
					faith = faith:wsl_slavic_west
					faith = faith:wsl_slavic_south
				}
			}
			debug_log = "  [PASS] Faith: [ROOT.GetFaith.GetName]"
		}
		else = {
			debug_log = "  [FAIL] Faith: [ROOT.GetFaith.GetName] — need East/West/South"
		}

		if = {
			limit = { piety >= 2000 }
			debug_log = "  [PASS] Piety >= 2000"
		}
		else = {
			debug_log = "  [FAIL] Piety < 2000"
		}

		if = {
			limit = { prestige >= 1500 }
			debug_log = "  [PASS] Prestige >= 1500"
		}
		else = {
			debug_log = "  [FAIL] Prestige < 1500"
		}

		debug_log = "  [TODO] Holy sites check: placeholder triggers"
		debug_log = "=== WSL REFORM CHECK: END ==="
	}

	option = { name = wsl_debug_ok }
}

	hidden = yes

	immediate = {

		debug_log = "==================================================================="

		set_variable = { name = wsl_snap_base    value = 0 }

		set_variable = { name = wsl_snap_west    value = 0 }
		set_variable = { name = wsl_snap_south   value = 0 }
		set_variable = { name = wsl_snap_unified value = 0 }

		every_county = {
			limit = { faith = faith:wsl_slavic_base }
			root = { change_variable = { name = wsl_snap_base add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_east }
			root = { change_variable = { name = wsl_snap_east add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_west }
			root = { change_variable = { name = wsl_snap_west add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_south }
			root = { change_variable = { name = wsl_snap_south add = 1 } }
		}
		every_county = {
			limit = { faith = faith:wsl_slavic_unified }
			root = { change_variable = { name = wsl_snap_unified add = 1 } }
		}

		debug_log = "FAITH: base=[root.MakeScope.Var('wsl_snap_base').GetValue|0] east=[root.MakeScope.Var('wsl_snap_east').GetValue|0] west=[root.MakeScope.Var('wsl_snap_west').GetValue|0] south=[root.MakeScope.Var('wsl_snap_south').GetValue|0] unified=[root.MakeScope.Var('wsl_snap_unified').GetValue|0]"
		debug_log = "PLAYER: [ROOT.GetName] | faith=[ROOT.GetFaith.GetName]"

			limit = {

		else = {
			debug_log = "UNIFIED: not yet"

		remove_variable = wsl_snap_base
		remove_variable = wsl_snap_east
		remove_variable = wsl_snap_west
