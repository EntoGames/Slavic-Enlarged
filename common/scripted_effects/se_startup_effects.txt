########################################
# WSL — przypisanie wiar słowiańskich w 867
# CEL: w dacie startu se_slavic_base NIE WYSTĘPUJE na mapie.
#      Wszystkie hrabstwa religion:slavic_religion trafiają
#      do jednego z 3 odłamów (east/west/south) na podstawie
#      regionu geograficznego, z dodatkową korektą po pillarze
#      kulturowym dla obszarów mieszanych.
########################################

se_assign_starting_faiths_effect = {

	# === KROK 1: ROZDZIAŁ GEOGRAFICZNY (podstawowy) ===

	# --- RODNOWIERSTWO ZACHODNIE ---
	# Połabie, Pomorze, zachodnia Germania (Obodryci, Lucice, Wieleci, Rani, Drzewianie, Serbowie Łużyccy)
	every_county_in_region = {
		region = world_europe_west_germania
		limit = {
			faith = { religion = religion:slavic_religion }
		}
		set_county_faith = faith:se_slavic_west
	}

	# --- RODNOWIERSTWO POŁUDNIOWE ---
	# Bałkany — Serbowie, Chorwaci, Pannonscy Słowianie itd.
	every_county_in_region = {
		region = world_europe_south_east
		limit = {
			faith = { religion = religion:slavic_religion }
		}
		set_county_faith = faith:se_slavic_south
	}

	# --- EUROPA WSCHODNIA: KOREKTA PO PILLARZE ---
	# Polacy, Czesi i inni zachodni Słowianie żyjący w world_europe_east
	# najpierw oznaczamy jako zachodnich, ZANIM ogólna reguła wschodnia ich złapie.
	every_county_in_region = {
		region = world_europe_east
		limit = {
			faith = { religion = religion:slavic_religion }
			holder = {
				culture = { has_cultural_pillar = heritage_west_slavic }
			}
		}
		set_county_faith = faith:se_slavic_west
	}

	# Resztę Europy Wschodniej (Rus, Krywicze, Polanie kijowscy itd.) → wschodnie
	every_county_in_region = {
		region = world_europe_east
		limit = {
			faith = { religion = religion:slavic_religion }
		}
		set_county_faith = faith:se_slavic_east
	}

	# === KROK 2: KOŃCOWE ZABEZPIECZENIE ===
	# Każde hrabstwo, które nadal ma religion:slavic_religion (poza powyższymi regionami)
	# zostaje przypisane wg pillara, a w ostateczności jako wschodnie.
	# Po tym kroku NIKT nie powinien już mieć se_slavic_base.
	every_county = {
		limit = {
			faith = { religion = religion:slavic_religion }
		}
		if = {
			limit = {
				holder = { culture = { has_cultural_pillar = heritage_west_slavic } }
			}
			set_county_faith = faith:se_slavic_west
		}
		else_if = {
			limit = {
				holder = { culture = { has_cultural_pillar = heritage_south_slavic } }
			}
			set_county_faith = faith:se_slavic_south
		}
		else = {
			# domyślnie wschodnie (największa grupa słowiańska)
			set_county_faith = faith:se_slavic_east
		}
	}

	# === KROK 3: KONWERSJA WŁADCÓW ===
	# Każdy władca z religion:slavic_religion przyjmuje wiarę ze swojego hrabstwa stołecznego
	every_ruler = {
		limit = {
			faith = { religion = religion:slavic_religion }
		}
		set_character_faith = capital_county.title_province.faith
	}

	# === KROK 4: AWARYJNA KONWERSJA POZOSTAŁYCH POSTACI ===
	# Wszystkie postacie (także BEZZIEMNI dworzanie/rodzina/pula) które wciąż
	# mają se_slavic_base LUB vanilla religię slavic_religion (KROK 3 łapie
	# tylko władców — bezziemni z vanilla wiarą zostawali nietknięci).
	every_living_character = {
		limit = {
			OR = {
				faith = faith:se_slavic_base
				faith = { religion = religion:slavic_religion }
			}
		}
		se_set_slavic_faith_by_pillar_effect = yes
	}
}

########################################
# HELPER: fallback wiary słowiańskiej wg pillara kulturowego
# Współdzielony przez startup (KROK 4) i periodyczny sweep leaku.
# Scope: character.
########################################
se_set_slavic_faith_by_pillar_effect = {
	if = {
		limit = { culture = { has_cultural_pillar = heritage_west_slavic } }
		set_character_faith = faith:se_slavic_west
	}
	else_if = {
		limit = { culture = { has_cultural_pillar = heritage_south_slavic } }
		set_character_faith = faith:se_slavic_south
	}
	else = {
		# domyślnie wschodnie (największa grupa słowiańska)
		set_character_faith = faith:se_slavic_east
	}
}

########################################
# FIX LEAKU BEZZIEMNYCH DWORZAN (EPIC 03 #25)
# CK3 generuje dworzan z DOMYŚLNYMI wartościami:
#   - wiara = vanilla slavic_religion (faith slavic_pagan, "Słowiańska prawda")
#            lub przejściowe se_slavic_base — w obu wypadkach NIE wiara moda
#   - kultura = vanilla kultura słowiańska (np. polish) zamiast
#            podzielonej kultury moda
# Nowym dworzanom CK3 nadaje wiarę gospodarza automatycznie, ale KULTURĘ
# losuje regionalnie — stąd okresowe "wycieki" vanilla kultury w dworze.
#
# se_normalize_courtier_to_host_effect — scope: DWORZANIN (bezziemny).
# Normalizuje SIEBIE do wiary/kultury gospodarza (host = właściciel dworu).
# Reużywany przez: sweep dworu (puls/start) ORAZ on_birthday (per postać,
# rozłożone w czasie — skraca okno widocznego leaku do urodzin dworzanina).
########################################
se_normalize_courtier_to_host_effect = {
	if = {
		limit = {
			is_landed = no
			exists = host
			# Tylko dwory słowiańskich gospodarzy
			host = { se_is_slavic_culture_trigger = yes }
		}
		# --- WIARA: leak (vanilla slavic_religion lub se_slavic_base) → wiara gospodarza ---
		if = {
			limit = {
				OR = {
					faith = faith:se_slavic_base
					faith = { religion = religion:slavic_religion }
				}
				NOT = { faith = host.faith }
				host = { se_character_has_slavic_faith_trigger = yes }
			}
			set_character_faith = host.faith
		}
		# --- KULTURA: słowiański dworzanin innej kultury niż gospodarz ---
		# Nie ruszamy obcych (Grecy, Niemcy itd.) — tylko leak słowiański.
		if = {
			limit = {
				se_is_slavic_culture_trigger = yes
				NOT = { culture = host.culture }
			}
			set_culture = host.culture
		}
	}
}

########################################
# Sweep CAŁEGO dworu jednego władcy. Scope: character (władca).
# Wołany: startowo per władca (czyści istniejące dwory od tury 0) ORAZ
# corocznie (yearly_playable_ruler_pulse — łapie nowo wygenerowanych).
########################################
se_fix_courtier_leaks_effect = {
	if = {
		limit = { se_is_slavic_culture_trigger = yes }

		# Defensywnie: sam władca nie powinien mieć se_slavic_base.
		if = {
			limit = { faith = faith:se_slavic_base }
			se_set_slavic_faith_by_pillar_effect = yes
		}

		every_courtier = {
			se_normalize_courtier_to_host_effect = yes
		}
	}
}
