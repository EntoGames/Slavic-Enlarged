# Chain into vanilla on_game_start without overriding its effect block.
# Vanilla on_game_start has its own 'effect' that we MUST NOT overwrite.
# Instead, we register a custom on_action that gets called from on_game_start.

on_game_start = {
	on_actions = {
		se_on_game_start
	}
}

se_on_game_start = {
	effect = {
		se_assign_starting_faiths_effect = yes
		se_assign_starting_cultures_effect = yes
		se_set_cultural_acceptance_effect = yes
		# Wyczyść leak vanilla wiary/kultury u istniejących dworzan od tury 0.
		# (musi iść PO przypisaniu wiar i kultur władcom)
		every_ruler = {
			se_fix_courtier_leaks_effect = yes
		}
	}
}

# Coroczny sweep dworu KAŻDEGO władcy: konwertuje wygenerowanych przez CK3
# dworzan z vanilla wiarą (slavic_religion) / se_slavic_base i vanilla kulturą
# na WIARĘ i KULTURĘ lokalnego władcy. (EPIC 03 #25 — fix leaku bezziemnych)
# root = władca (vanilla on_action, odpala się dla wszystkich landed AI+gracz).
yearly_playable_ruler_pulse = {
	effect = {
		se_fix_courtier_leaks_effect = yes
	}
}

# Per-dworzanin: każdy bezziemny normalizuje się do gospodarza w swoje
# urodziny. Rozłożone w czasie => skraca okno widocznego leaku kultury
# (puls władcy łapie raz w roku; urodziny — wcześniej dla części dworzan).
# root = postać obchodząca urodziny.
on_birthday = {
	on_actions = {
		se_on_birthday
	}
}

se_on_birthday = {
	effect = {
		se_normalize_courtier_to_host_effect = yes
	}
}

# Odpala się gdy jakikolwiek władca zreformuje wiarę.
# Root scope: faith (reformowana wiara). scope:reformer: postać reformatora.
on_faith_reformed = {
	on_actions = {
		se_on_slavic_faith_reformed
	}
}

se_on_slavic_faith_reformed = {
	effect = {
		if = {
			limit = {
				OR = {
					this = faith:se_slavic_east
					this = faith:se_slavic_west
					this = faith:se_slavic_south
				}
			}
			scope:reformer = {
				if = {
					limit = { se_controls_9_slavic_holy_sites_trigger = yes }
					trigger_event = { id = se_reform.0001 }	# nagroda
				}
				else_if = {
					limit = { se_controls_6_slavic_holy_sites_trigger = yes }
					trigger_event = { id = se_reform.0003 }	# stan zerowy
				}
				else = {
					trigger_event = { id = se_reform.0002 }	# kara
				}
			}
		}
	}
}
