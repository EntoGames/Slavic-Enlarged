# ─── Kultura ───────────────────────────────────────────────────

se_is_slavic_culture_trigger = {
	OR = {
		culture = { has_cultural_pillar = heritage_west_slavic }
		culture = { has_cultural_pillar = heritage_east_slavic }
		culture = { has_cultural_pillar = heritage_south_slavic }
	}
}

# ─── Wiara ─────────────────────────────────────────────────────

se_is_slavic_pagan_faith_trigger = {
	OR = {
		faith = faith:se_slavic_base
		faith = faith:se_slavic_east
		faith = faith:se_slavic_west
		faith = faith:se_slavic_south
	}
}

# Alias na character scope — convenience trigger dla eventów
se_character_has_slavic_faith_trigger = {
	OR = {
		faith = faith:se_slavic_base
		faith = faith:se_slavic_east
		faith = faith:se_slavic_west
		faith = faith:se_slavic_south
	}
}
