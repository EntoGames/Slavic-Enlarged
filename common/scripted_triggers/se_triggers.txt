se_is_slavic_culture_trigger = {
	OR = {
		culture = { has_cultural_pillar = heritage_west_slavic }
		culture = { has_cultural_pillar = heritage_east_slavic }
		culture = { has_cultural_pillar = heritage_south_slavic }
	}
}

se_is_slavic_pagan_faith_trigger = {
	OR = {
		faith = faith:se_slavic_base
		faith = faith:se_slavic_east
		faith = faith:se_slavic_west
		faith = faith:se_slavic_south
	}
}

# Progi kontroli słowiańskich miejsc świętych:
#   >= 9 → nagroda   (se_reform.0001)
#   >= 6 → stan zero (se_reform.0003)
#   <  6 → kara      (se_reform.0002)

se_controls_9_slavic_holy_sites_trigger = {
	calc_true_if = {
		amount = 9
		# wschodnie
		any_realm_county = { this = title:c_lviv }
		any_realm_county = { this = title:c_kiev }
		any_realm_county = { this = title:c_minsk }
		any_realm_county = { this = title:c_novgorod }
		any_realm_county = { this = title:c_polotsk }
		# zachodnie
		any_realm_county = { this = title:c_rugen }
		any_realm_county = { this = title:c_plocka }
		any_realm_county = { this = title:c_praha }
		any_realm_county = { this = title:c_legnica }
		any_realm_county = { this = title:c_mecklenburg }
		# południowe
		any_realm_county = { this = title:c_velbazhd }
		any_realm_county = { this = title:c_tolna }
		any_realm_county = { this = title:c_zachlumia }
		any_realm_county = { this = title:c_barlad }

	}
}
