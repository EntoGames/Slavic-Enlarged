wsl_pagan_slavs_threshold_trigger = {
	any_ruler = {
		count >= 5
		wsl_is_slavic_culture_trigger = yes
		wsl_is_slavic_pagan_faith_trigger = yes
		highest_held_title_tier >= tier_county
	}
}

wsl_is_slavic_culture_trigger = {
	OR = {
		culture = { has_cultural_pillar = heritage_west_slavic }
		culture = { has_cultural_pillar = heritage_east_slavic }
		culture = { has_cultural_pillar = heritage_south_slavic }
	}
}

wsl_is_slavic_pagan_faith_trigger = {
	OR = {
		faith = faith:wsl_slavic_base
		faith = faith:wsl_slavic_east
		faith = faith:wsl_slavic_west
		faith = faith:wsl_slavic_south
		faith = faith:wsl_slavic_unified
	}
}
